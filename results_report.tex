\begin{abstract}
This report presents a comprehensive analysis of Microsoft (MSFT) stock forecasting using multiple time series methods across different timeframes and experimental configurations. The study compares five forecasting methods—ARIMA, Chronos Base, FM (Foundation Model), Naive, and Sundial—across daily (1d) and hourly (1h) data frequencies. Three experimental setups were employed: train-less-year-linear, train-less-year-log, and train-restricted-years. Performance was evaluated using standard metrics including MSE, MAE, RMSE, and MAPE across various training periods and prediction horizons.
\end{abstract}

\section{Introduction}

\subsection{Timeframes}
The analysis encompasses two primary data frequencies:
\begin{itemize}
\item \textbf{Daily (1d)}: End-of-day closing prices for long-term trend analysis
\item \textbf{Hourly (1h)}: Intraday price movements for short-term forecasting
\end{itemize}

\subsection{Experimental Configurations}
Three distinct training approaches were implemented:
\begin{itemize}
\item \textbf{train-less-year-linear}: Linear scaling with training periods under one year
\item \textbf{train-less-year-log}: Logarithmic scaling with training periods under one year  
\item \textbf{train-restricted-years}: Training restricted to specific year ranges
\end{itemize}

\subsection{Forecasting Methods}
\begin{itemize}
\item \textbf{ARIMA}: AutoRegressive Integrated Moving Average, a classical statistical method
\item \textbf{Chronos Base}: Pre-trained time series foundation model from Amazon
\item \textbf{FM}: Foundation Model approach for time series forecasting
\item \textbf{Naive}: Simple baseline using last observed value
\item \textbf{Sundial}: Specialized time series forecasting model
\end{itemize}

\subsection{Evaluation Metrics}
Performance was assessed using:
\begin{itemize}
\item \textbf{MSE}: Mean Squared Error
\item \textbf{MAE}: Mean Absolute Error  
\item \textbf{RMSE}: Root Mean Squared Error
\item \textbf{MAPE}: Mean Absolute Percentage Error
\end{itemize}

\section{Comparative Analysis}

\subsection{Method Performance Comparison}

\subsubsection{Daily Forecasting}
Based on the comprehensive analysis across all experiments:

\begin{itemize}
\item \textbf{Best Overall Performance}: Chronos Base demonstrates consistent performance with MAPE typically around 1.28-1.31\% across all training configurations
\item \textbf{Most Variable}: ARIMA shows high variability, with performance heavily dependent on training period (MAPE ranging from 1.22\% to 5.97\%)
\item \textbf{Baseline Performance}: Naive method provides a reasonable baseline with stable but suboptimal performance
\item \textbf{Training Sensitivity}: Classical methods (ARIMA) show higher sensitivity to training duration compared to foundation models
\end{itemize}

\subsubsection{Hourly Forecasting}
Hourly predictions present different challenges:

\begin{itemize}
\item \textbf{Increased Volatility}: All methods show higher error rates for hourly predictions compared to daily
\item \textbf{Foundation Model Advantage}: Chronos Base and FM maintain relative stability in the higher frequency domain
\item \textbf{ARIMA Challenges}: Traditional statistical methods struggle with intraday noise and microstructure effects
\end{itemize}

\subsection{Training Configuration Impact}

\subsubsection{Linear vs Logarithmic Scaling}
\begin{itemize}
\item \textbf{Linear Configuration}: Better for shorter training periods, shows clear improvement patterns
\item \textbf{Logarithmic Configuration}: More stable performance across varying training sizes
\item \textbf{Optimal Training}: 75-100 days appears optimal for most methods in daily timeframe
\end{itemize}

\subsubsection{Restricted Years}
\begin{itemize}
\item \textbf{Data Quality Impact}: Performance varies significantly based on the specific years included
\item \textbf{Market Regime Sensitivity}: Methods show different sensitivities to market conditions in different periods
\end{itemize}

\section{Conclusions and Recommendations}

\subsection{Key Findings}

\begin{enumerate}
\item \textbf{Foundation Models Superiority}: Chronos Base and FM consistently outperform traditional methods across most configurations
\item \textbf{Training Period Optimization}: Optimal training periods exist for each method, typically 75-100 days for daily data
\item \textbf{Frequency Dependency}: Method performance varies significantly between daily and hourly timeframes
\item \textbf{Configuration Sensitivity}: Training configuration impacts classical methods more than foundation models
\end{enumerate}

\subsection{Practical Recommendations}

\begin{itemize}
\item \textbf{For Daily Forecasting}: Use Chronos Base with 75-100 days of training data
\item \textbf{For Hourly Forecasting}: Foundation models (Chronos Base or FM) are strongly recommended
\item \textbf{Resource Constraints}: Naive method provides acceptable baseline performance with minimal computational requirements
\item \textbf{Model Selection}: Consider Chronos Base for consistent performance across different market conditions
\end{itemize}

\subsection{Future Research Directions}

\begin{itemize}
\item \textbf{Hybrid Approaches}: Combining foundation models with traditional statistical methods
\item \textbf{Multi-horizon Optimization}: Joint optimization across different prediction horizons
\item \textbf{Market Regime Adaptation}: Dynamic model selection based on market conditions
\item \textbf{Real-time Implementation}: Deployment considerations for production environments
\end{itemize}
